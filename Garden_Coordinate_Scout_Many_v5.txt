local function tryCall(name, ...)
    local fn
    if type(getgenv) == "function" then
        local env = getgenv()
        if type(env[name]) == "function" then fn = env[name] end
    end
    if not fn and type(_G[name]) == "function" then fn = _G[name] end
    if not fn and name == "writefile" and type(writefile) == "function" then fn = writefile end
    if not fn and name == "readfile" and type(readfile) == "function" then fn = readfile end
    if not fn and name == "isfile" and type(isfile) == "function" then fn = isfile end
    if not fn and name == "setclipboard" and type(setclipboard) == "function" then fn = setclipboard end
    if not fn and name == "toclipboard" and type(toclipboard) == "function" then fn = toclipboard end
    if not fn then return false, nil end
    return pcall(fn, ...)
end

local function copyText(text)
    local ok = tryCall("setclipboard", text)
    if ok then return true end
    ok = tryCall("toclipboard", text)
    return ok
end

local function encodeMarks(marks)
    local data = {}
    for i, p in ipairs(marks) do
        data[i] = {x = p.X, y = p.Y, z = p.Z}
    end
    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, data)
    return ok and encoded or nil
end

local function saveMarks(marks)
    local encoded = encodeMarks(marks)
    if not encoded then return false, "JSON encode failed" end
    local ok = tryCall("writefile", SAVE_FILE, encoded)
    return ok, ok and nil or "writefile unavailable"
end

local function loadMarks()
    local exists = tryCall("isfile", SAVE_FILE)
    if not exists then return {} end
    local ok, contents = tryCall("readfile", SAVE_FILE)
    if not ok or type(contents) ~= "string" then return {} end
    local decodedOk, decoded = pcall(HttpService.JSONDecode, HttpService, contents)
    if not decodedOk or type(decoded) ~= "table" then return {} end

    local marks = {}
    for _, v in ipairs(decoded) do
        if #marks >= MAX_MARKS then break end
        if type(v) == "table" and tonumber(v.x) and tonumber(v.y) and tonumber(v.z) then
            marks[#marks + 1] = Vector3.new(tonumber(v.x), tonumber(v.y), tonumber(v.z))
        end
    end
    return marks
end

local Marks = loadMarks()

local MarkerObjects = {}

local function makeMarker(position, index)
    local part = Instance.new("Part")
    part.Name = "Mark_" .. index
    part.Size = Vector3.new(0.9, 0.14, 0.9)
    part.Position = position + Vector3.new(0, 0.18, 0)
    part.Anchored = true
    part.CanCollide = false
    part.CanTouch = false
    part.CanQuery = false
    part.Material = Enum.Material.Neon
    part.Color = Color3.fromRGB(80, 255, 135)
    part.Transparency = 0.12
    part.Parent = MarkerFolder
    return part
end

local function rebuildMarkers()
    for _, marker in ipairs(MarkerObjects) do
        if marker and marker.Parent then marker:Destroy() end
    end
    table.clear(MarkerObjects)
    for i, point in ipairs(Marks) do
        MarkerObjects[i] = makeMarker(point, i)
    end
end

local markersVisible = true

local function refreshMarkers()
    for _, marker in ipairs(MarkerObjects) do
        if marker and marker.Parent then
            marker.Transparency = markersVisible and 0.12 or 1
        end
    end
end

rebuildMarkers()

local function getCenterHit()
    local camera = workspace.CurrentCamera
    if not camera then return nil end

    local viewport = camera.ViewportSize
    local ray = camera:ViewportPointToRay(viewport.X * 0.5, viewport.Y * 0.5, 0)

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.IgnoreWater = false
    params.FilterDescendantsInstances = {LocalPlayer.Character, MarkerFolder}

    local result = workspace:Raycast(ray.Origin, ray.Direction * RAY_LENGTH, params)
    return result and result.Position or nil
end

local function vectorString(point)
    return string.format("Vector3.new(%.9f, %.9f, %.9f)", point.X, point.Y, point.Z)
end

local function closestDistance(point)
    local closest = math.huge
    for _, p in ipairs(Marks) do
        local d = (p - point).Magnitude
        if d < closest then closest = d end
    end
    return closest
end

    Count.Text = ("Saved marks: %d / %d"):format(#Marks, MAX_MARKS)

MarkButton.Activated:Connect(function()
    local point = getCenterHit()

    if not point then
        Status.Text = "Status: NO SURFACE UNDER CROSSHAIR"
        Status.TextColor3 = Color3.fromRGB(255, 150, 120)
        return
    end

    if #Marks >= MAX_MARKS then
        Status.Text = ("Status: MAX %d MARKS REACHED"):format(MAX_MARKS)
        Status.TextColor3 = Color3.fromRGB(255, 205, 110)
        return
    end

    local closest = closestDistance(point)

    if closest < MIN_MARK_DISTANCE then
        Status.Text = ("Status: TOO CLOSE TO EXISTING MARK (%.2f)"):format(closest)
        Status.TextColor3 = Color3.fromRGB(255, 205, 110)
        return
    end

    Marks[#Marks + 1] = point
    local index = #Marks
    MarkerObjects[index] = makeMarker(point, index)

    saveMarks(Marks)
    updateCount()

    Status.Text = ("Status: MARK %d SAVED"):format(index)
    Status.TextColor3 = Color3.fromRGB(120, 225, 145)

    print(("[Garden Scout v5] Mark %d = %s"):format(index, vectorString(point)))
end)

local dragging = false
local dragStart
local startPosition

local DragHandle = Instance.new("Frame")
DragHandle.Name = "DragHandle"
DragHandle.Position = UDim2.fromOffset(0, 0)
DragHandle.Size = UDim2.new(1, 0, 0, 42)
DragHandle.BackgroundTransparency = 1
DragHandle.Active = true
DragHandle.Parent = Main

DragHandle.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = Main.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end

    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
        local delta = input.Position - dragStart

        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
